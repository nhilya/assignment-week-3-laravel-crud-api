        ```
        public function rules(): array
        {
            if ($this->is('*/login')) {
                return [
                    'email' => 'required|email',
                    'password' => 'required',
                ];

            return [
                'name' => 'required',
                'email' => 'required|email',
                'password' => 'required',
            ];
        }
        ```
